Core Modules

Server Module

Main server implementation for FTP/SFTP protocols

// Example: Creating a new server
const { createServer } = require('./src/server');

const server = createServer({
    protocol: 'sftp',
    port: 22,
    host: '0.0.0.0'
});

server.start();

Database Module

SQLite database operations for configuration management

// Example: Database operations
const { Database } = require('./src/database');

const db = new Database('config.db');

// Create user
await db.createUser({
    username: 'newuser',
    password: 'securepassword',
    permissions: ['read', 'write']
});

// Get user
const user = await db.getUser('newuser');

Authentication Module

User authentication handling for both password and public key

// Example: Authentication
const { authenticate } = require('./src/auth');

// Password authentication
const result = await authenticate.password('username', 'password');

// Public key authentication
const result = await authenticate.publicKey('username', publicKey);

Configuration API

User Management

// User CRUD operations
const { UserManager } = require('./src/managers/user');

const userManager = new UserManager(db);

// Create user
await userManager.create({
    username: 'user1',
    password: 'pass123',
    guiAccess: true
});

// Update user
await userManager.update('user1', {
    password: 'newpass456'
});

// Delete user
await userManager.delete('user1');

// List users
const users = await userManager.list();

Server Management

// Server configuration operations
const { ServerManager } = require('./src/managers/server');

const serverManager = new ServerManager(db);

// Create listener
await serverManager.createListener({
    protocol: 'ftp',
    port: 2121,
    host: '127.0.0.1',
    enabled: true
});

// Update listener
await serverManager.updateListener(1, {
    port: 2122
});

// Delete listener
await serverManager.deleteListener(1);

Events API

Server Events

// Server event handling
const { EventEmitter } = require('events');

class Server extends EventEmitter {
    // Connection events
    on('connection', (client) => {
        console.log('New client connected:', client.address);
    });

    on('disconnect', (client) => {
        console.log('Client disconnected:', client.address);
    });

    on('error', (error) => {
        console.error('Server error:', error);
    });
}

File Operation Events

// File operation monitoring
server.on('upload', (data) => {
    console.log('File uploaded:', data.filename);
});

server.on('download', (data) => {
    console.log('File downloaded:', data.filename);
});

server.on('delete', (data) => {
    console.log('File deleted:', data.filename);
});

Plugin System

Custom Plugins

// Creating custom plugins
const { Plugin } = require('./src/plugin');

class CustomPlugin extends Plugin {
    constructor() {
        super('custom-plugin');
    }

    async onLoad() {
        console.log('Plugin loaded');
    }

    async onUnload() {
        console.log('Plugin unloaded');
    }

    async onFileUpload(data) {
        // Custom file upload handling
    }
}

// Register plugin
server.registerPlugin(new CustomPlugin());
